在构建分布式锁时，可以使用redis作为缓存中间件来实现，而redis的实现又有下面多种方法。

### setnx
setnx+px可以实现分布式锁：
- 某个key存在，则setnx失败，表示加锁失败
- 如果某个key不存在，则setnx成功，并且有一定的过期时间，表示加锁成功，并且锁会在一定时限之后自动解锁

### lua脚本
```
	rsp, err := rl.client.Eval("if redis.call('get', KEYS[1]) == ARGV[1] then return redis.call('expire', KEYS[1], ARGV[2]) else return 0 end", []string{rl.lockKey}, rl.uuid, int64(rl.ttl/time.Second)).Result()
```
这个 Lua 脚本的功能是：
- 检查 Redis 键 KEYS[1] 的值是否等于 ARGV[1]。
- 如果相等，则设置该键的过期时间为 ARGV[2] 秒。
- 如果不相等，则返回 0。
### redis集群的redlock
如果redis是一个集群，上面两种方法则需要保证它们都对同一个节点进行操作。
而redlock算法，则是基于节点的大多数同意这一原则，进行锁操作。
```
package main

import (
    "github.com/go-redis/redis/v8"
    "context"
    "time"
    "crypto/rand"
    "encoding/hex"
    "fmt"
)

func randomID() string {
    b := make([]byte, 16)
    rand.Read(b)
    return hex.EncodeToString(b)
}

func acquireLock(clients []*redis.Client, lockKey string, ttl time.Duration) (bool, string) {
    value := randomID()
    majority := (len(clients) / 2) + 1
    acquired := 0

    for _, client := range clients {
        res, err := client.SetNX(context.Background(), lockKey, value, ttl).Result()
        if err == nil && res {
            acquired++
        }
    }

    return acquired >= majority, value
}

func releaseLock(clients []*redis.Client, lockKey, value string) {
    for _, client := range clients {
        client.Eval(context.Background(), `
            if redis.call("get", KEYS[1]) == ARGV[1] then
                return redis.call("del", KEYS[1])
            else
                return 0
            end
        `, []string{lockKey}, value)
    }
}

func main() {
    // 假设有5个Redis客户端
    clients := []*redis.Client{
        redis.NewClient(&redis.Options{Addr: "localhost:6379"}),
        redis.NewClient(&redis.Options{Addr: "localhost:6380"}),
        redis.NewClient(&redis.Options{Addr: "localhost:6381"}),
        redis.NewClient(&redis.Options{Addr: "localhost:6382"}),
        redis.NewClient(&redis.Options{Addr: "localhost:6383"}),
    }

    lockKey := "myLock"
    ttl := 10 * time.Second

    success, value := acquireLock(clients, lockKey, ttl)
    if success {
        fmt.Println("Lock acquired")
        // 执行需要保护的操作

        releaseLock(clients, lockKey, value)
        fmt.Println("Lock released")
    } else {
        fmt.Println("Failed to acquire lock")
    }
}
```
